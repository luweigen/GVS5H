   size = 2*OFFSET + 1
   mask = (1 << size) - 1
4. Initialize dp_reach = [ [0]*(limit+1) for _ in range(2) ]
   dp_nonempty = [ [0]*(limit+1) for _ in range(2) ]
   Set dp_reach[0][1] = 1 << OFFSET  (bit for sum 0)
   dp_nonempty remains zero.
5. For each x in nums:
   - Create next_reach = [row[:] for row in dp_reach]  (skip)
   - Create next_nonempty = [row[:] for row in dp_nonempty] (skip)
   - For par in [0,1]:
       for prod in range(limit+1):
           bs = dp_reach[par][prod]
           if bs == 0: continue
           new_par = 1 - par
           new_prod = prod * x
           if new_prod > limit: continue
           if par == 0:
               new_bs = (bs << x) & mask
           else:
               new_bs = (bs >> x) & mask
           next_reach[new_par][new_prod] |= new_bs
